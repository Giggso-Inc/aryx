Env Vars to Activate aryx-o

# Edition flag
ARYX_EDITION=aryx-o

# Service 1 — Oracle Graph Studio (replaces FalkorDB)
ARYX_ADAPTER_GRAPH_READER=aryx.adapters.oracle.graph:OracleGraphReader
ARYX_ADAPTER_GRAPH_STORE=aryx.adapters.oracle.graph:OracleGraphStore

# Service 2 — Oracle ADB 23ai (replaces Postgres + Select AI for Ask)
ARYX_ADAPTER_RELATIONAL=aryx.adapters.oracle.rdb:OracleRelationalAdapter

# Service 3 — OCI GenAI (replaces Ollama/Anthropic/OpenAI)
ARYX_ADAPTER_LLM=aryx.adapters.oracle.genai:OracleGenAiLlm

# OCI Document Understanding (replaces pymupdf + tesseract)
ARYX_DOCUMENT_CONNECTOR=oci

# ReasonerPort — Oracle PGX OWL 2 RL (replaces Python rule loop)
ARYX_ADAPTER_REASONER=aryx.adapters.oracle.pgx:OraclePgxReasoner

# ⚠ All six adapter classes above are the implementation gap —
